🚀 Getting Started

-- Load Milkyway UI (place at top of your script)
local Milkyway = loadstring(game:HttpGet("https://milky.vercel.app/milkyway.lua"))()

-- Create main window (automatically appears with left profile panel)
local mainTab = Milkyway:CreateTab("🏠", "Home")
local settingsTab = Milkyway:CreateTab("⚙️", "Settings")

-- Button example
Milkyway:CreateButton(mainTab, "Click Me", "🔘", function()
    Milkyway:Notify("System", "Button clicked!", "success", 2)
end)

-- Toggle
Milkyway:CreateToggle(mainTab, "Enable Feature", true, function(state)
    print("Feature toggled:", state)
end)

-- Slider
Milkyway:CreateSlider(mainTab, "Volume", 0, 100, 50, "%", function(value)
    game:GetService("SoundService").MasterVolume = value / 100
end)

-- Dropdown
Milkyway:CreateDropdown(mainTab, "Select Mode", {"Auto", "Semi", "Manual"}, function(selected)
    print("Mode:", selected)
end)

-- Theme selector (adds theme buttons automatically)
Milkyway:CreateThemeSelector(settingsTab)

-- Notification (Discord-style under left avatar)
Milkyway:Notify("Milkyway", "Library ready!", "info", 3)

🖥️ Real‑World Example

local Milky = loadstring(game:HttpGet("https://milky.vercel.app/milkyway.lua"))()

local main = Milky:CreateTab("🏠", "Main")
local cfg = Milky:CreateTab("⚙️", "Config")

Milky:CreateSection(main, "Player")
Milky:CreateButton(main, "Kill All", "💀", function()
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr.Character then plr.Character.Humanoid.Health = 0 end
    end
    Milky:Notify("Combat", "Executed!", "error", 2)
end)

local flyEnabled = false
Milky:CreateToggle(main, "Fly Mode", false, function(state)
    flyEnabled = state
    -- fly code here
end)

Milky:CreateSlider(main, "WalkSpeed", 16, 120, 16, "", function(val)
    game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = val
end)

Milky:CreateThemeSelector(cfg)
Milky:Notify("Milkyway", "Example script ready", "success", 2)
